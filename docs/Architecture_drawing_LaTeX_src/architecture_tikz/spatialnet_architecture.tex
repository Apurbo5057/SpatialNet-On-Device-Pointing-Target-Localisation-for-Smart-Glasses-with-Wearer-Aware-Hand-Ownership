% \slab[front-face options]{name}{left x}{centre y}{width}{height}{depth}{colour}
% front face = spatial H x W, extrusion = channels. Defines (name-w/e/n/s/nw/sw).
\newcommand{\slab}[8][]{%
  \pgfmathsetmacro{\sx}{#7*\EX}\pgfmathsetmacro{\sy}{#7*\EY}%
  \pgfmathsetmacro{\xr}{#3+#5}\pgfmathsetmacro{\yb}{#4-#6/2}\pgfmathsetmacro{\yt}{#4+#6/2}%
  \filldraw[fill=#8!45!white, draw=#8!50!black, line width=.4pt, line join=round]
    (#3,\yt) -- ++(\sx,\sy) -- ++(#5,0) -- (\xr,\yt) -- cycle;
  \filldraw[fill=#8!70!black, draw=#8!50!black, line width=.4pt, line join=round]
    (\xr,\yb) -- ++(\sx,\sy) -- ++(0,#6) -- (\xr,\yt) -- cycle;
  \filldraw[fill=#8, draw=#8!50!black, line width=.4pt, #1] (#3,\yb) rectangle (\xr,\yt);
  \coordinate (#2-w) at (#3,#4);
  \coordinate (#2-e) at (\xr+\sx,#4);
  \coordinate (#2-n) at (#3+#5/2+\sx,\yt+\sy);
  \coordinate (#2-s) at (#3+#5/2,\yb);
  \coordinate (#2-nw) at (#3,\yt);
  \coordinate (#2-sw) at (#3,\yb);
}
% n x n grid on a square face: {left x}{bottom y}{size}{n}{style}
\newcommand{\facegrid}[5]{%
  \pgfmathsetmacro{\gst}{#3/#4}%
  \begin{scope}[shift={(#1,#2)}]\draw[#5, step=\gst] (0,0) grid (#3-0.001,#3-0.001);\end{scope}%
}
% feature tensor C x n x n: slab + n x n cell grid on the front face + channel slices on the extruded faces
% {name}{left x}{centre y}{face size}{depth}{colour}{n}{slices}
\newcommand{\tensor}[8]{%
  \slab{#1}{#2}{#3}{#4}{#4}{#5}{#6}%
  \facegrid{#2}{#3-#4/2}{#4}{#7}{white, opacity=.55, line width=.22pt}%
  \pgfmathsetmacro{\sx}{#5*\EX}\pgfmathsetmacro{\sy}{#5*\EY}%
  \pgfmathsetmacro{\xr}{#2+#4}\pgfmathsetmacro{\yb}{#3-#4/2}\pgfmathsetmacro{\yt}{#3+#4/2}%
  \foreach \q in {1,...,#8} {%
    \pgfmathsetmacro{\fr}{\q/(#8+1)}%
    \draw[#6!50!black, opacity=.45, line width=.25pt]
      (#2+\fr*\sx,\yt+\fr*\sy) -- (\xr+\fr*\sx,\yt+\fr*\sy) -- (\xr+\fr*\sx,\yb+\fr*\sy);%
  }%
}
% slab whose front face is an image: {name}{left x}{centre y}{size}{depth}{side colour}{file}
\newcommand{\imgslab}[7]{%
  \slab[fill=none]{#1}{#2}{#3}{#4}{#4}{#5}{#6}%
  \node[anchor=south west, inner sep=0pt] at (#2,#3-#4/2) {\includegraphics[width=#4cm,height=#4cm]{#7}};%
  \draw[#6!50!black, line width=.4pt] (#2,#3-#4/2) rectangle ++(#4,#4);%
}
% PlotNeuralNet-style block: thin front face (channels), large oblique side face (H x W) with an n x n grid.
% {name}{left x}{centre y}{thickness}{height}{depth}{colour}{slices}{grid n}. Defines (name-w/e/n/s).
\newcommand{\pnbox}[9]{%
  \pgfmathsetmacro{\pdx}{#6*\EX}\pgfmathsetmacro{\pdy}{#6*\EY}%
  \pgfmathsetmacro{\pst}{#4/#8}\pgfmathsetmacro{\pyb}{#3-#5/2}\pgfmathsetmacro{\pyt}{#3+#5/2}%
  \foreach \q in {1,...,#8} {%
    \pgfmathsetmacro{\pxa}{#2+(\q-1)*\pst}\pgfmathsetmacro{\pxb}{#2+\q*\pst}%
    \filldraw[fill=#7!40!white, draw=#7!45!black, line width=.35pt, line join=round]
      (\pxa,\pyt) -- ++(\pdx,\pdy) -- (\pxb+\pdx,\pyt+\pdy) -- (\pxb,\pyt) -- cycle;
    \filldraw[fill=#7!88!black, draw=#7!45!black, line width=.35pt, line join=round]
      (\pxb,\pyb) -- ++(\pdx,\pdy) -- (\pxb+\pdx,\pyt+\pdy) -- (\pxb,\pyt) -- cycle;
    \filldraw[fill=#7!62!white, draw=#7!45!black, line width=.35pt] (\pxa,\pyb) rectangle (\pxb,\pyt);
  }%
  \pgfmathsetmacro{\pxr}{#2+#4}%
  \foreach \q in {1,...,\numexpr#9-1\relax} {%
    \pgfmathsetmacro{\fr}{\q/#9}%
    \draw[white, opacity=.32, line width=.2pt] (\pxr+\fr*\pdx,\pyb+\fr*\pdy) -- (\pxr+\fr*\pdx,\pyt+\fr*\pdy);
    \draw[white, opacity=.32, line width=.2pt] (\pxr,\pyb+\fr*#5) -- (\pxr+\pdx,\pyb+\fr*#5+\pdy);
  }%
  \coordinate (#1-w) at (#2,#3);
  \coordinate (#1-e) at (\pxr+\pdx,#3);
  \coordinate (#1-n) at (#2+#4/2+\pdx,\pyt+\pdy);
  \coordinate (#1-s) at (#2+#4/2,\pyb);
}
% image lying on the oblique plane (PlotNeuralNet style): {name}{front x}{centre y}{size}{file}
\newcommand{\imgplane}[5]{%
  \pgfmathsetmacro{\pdx}{#4*\EX}\pgfmathsetmacro{\pdy}{#4*\EY}%
  \fill[black, opacity=.12] (#2+0.06,#3-#4/2-0.06) -- ++(\pdx,\pdy) -- ++(0,#4) -- (#2+0.06,#3+#4/2-0.06) -- cycle;
  \begin{scope}[cm={\EX,\EY,0,1,(#2,#3-#4/2)}]
    \node[anchor=south west, inner sep=0pt, transform shape] at (0,0) {\includegraphics[width=#4cm,height=#4cm]{#5}};
    \draw[gray!50!black, line width=.5pt, line join=round] (0,0) rectangle (#4,#4);
  \end{scope}
  \coordinate (#1-w) at (#2,#3);
  \coordinate (#1-e) at (#2+\pdx,#3);
}
% small fully connected MLP icon, three layers.
% \mlpH{left x}{centre y}{layer gap}{node gap}{colour}{n1}{n2}{n3}: flows left -> right
% \mlpV{centre x}{top y}{layer gap}{node gap}{colour}{n1}{n2}{n3}: flows top -> bottom
\newcommand{\mlpnode}[2]{\filldraw[fill=white, draw=#2, line width=.7pt] #1 circle (\mlpr);}
\def\mlpr{0.065}\def\mlppad{0.055}
\newcommand{\mlpH}[8]{%
  \foreach \a/\na/\nb in {0/#6/#7, 1/#7/#8}
    \foreach \i in {1,...,\na} \foreach \j in {1,...,\nb}
      \draw[#5!45!white, line width=.3pt] (#1+\a*#3,{#2+((\na+1)/2-\i)*#4}) -- (#1+\a*#3+#3,{#2+((\nb+1)/2-\j)*#4});
  \foreach \a/\na in {0/#6, 1/#7, 2/#8} {
    \draw[#5!55!white, line width=.6pt] ({#1+\a*#3-\mlpr-\mlppad},{#2-(\na-1)/2*#4-\mlpr-\mlppad}) rectangle ({#1+\a*#3+\mlpr+\mlppad},{#2+(\na-1)/2*#4+\mlpr+\mlppad});
    \foreach \i in {1,...,\na} {\mlpnode{(#1+\a*#3,{#2+((\na+1)/2-\i)*#4})}{#5}}
  }
}
\newcommand{\mlpV}[8]{%
  \foreach \a/\na/\nb in {0/#6/#7, 1/#7/#8}
    \foreach \i in {1,...,\na} \foreach \j in {1,...,\nb}
      \draw[#5!45!white, line width=.3pt] ({#1+(\i-(\na+1)/2)*#4},#2-\a*#3) -- ({#1+(\j-(\nb+1)/2)*#4},#2-\a*#3-#3);
  \foreach \a/\na in {0/#6, 1/#7, 2/#8} {
    \draw[#5!55!white, line width=.6pt] ({#1-(\na-1)/2*#4-\mlpr-\mlppad},#2-\a*#3-\mlpr-\mlppad) rectangle ({#1+(\na-1)/2*#4+\mlpr+\mlppad},#2-\a*#3+\mlpr+\mlppad);
    \foreach \i in {1,...,\na} {\mlpnode{({#1+(\i-(\na+1)/2)*#4},#2-\a*#3)}{#5}}
  }
}

}

}
